// ══ 月餅訂購系統 - Apps Script v2 ══ // 試算表為唯一資料來源，支援讀取、新增、更新、軟刪除 const SHEET_NAME = '訂單'; const HEADERS = [ '訂單ID', '訂購人', '取貨日期', '付款狀態', '是否出貨', '總金額', '商品明細', '備註', '建立時間', '已刪除' ]; // ── GET：讀取所有訂單 ── function doGet(e) { try { const sheet = getOrCreateSheet(); const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return jsonResponse({ success: true, orders: [] }); const orders = []; for (let i = 1; i < rows.length; i++) { const r = rows[i]; if (r[9] === true || r[9] === 'TRUE' || r[9] === '已刪除') continue; // 軟刪除跳過 if (!r[0]) continue; // 空列跳過 // 商品明細還原 const itemLines = String(r[6]).split('\n').filter(Boolean); const items = itemLines.map(line => parseItemLine(line)); orders.push({ id: r[0], name: r[1], date: r[2] instanceof Date ? r[2].toISOString().slice(0,10) : String(r[2]), pay: r[3] === '已付款' ? 'paid' : 'unpaid', shipped: r[4] === '已出貨', total: r[5], items, note: r[7] || '', createdAt: r[8] instanceof Date ? r[8].toISOString() : String(r[8]), }); } return jsonResponse({ success: true, orders }); } catch(err) { return jsonResponse({ success: false, error: err.message }); } } // 商品明細文字還原成物件（for 顯示用） function parseItemLine(line) { // 解析數量，例如「×3」 const qtyMatch = line.match(/×(\d+)/); const qty = qtyMatch ? parseInt(qtyMatch[1]) : 1; if (line.includes('鳳梨酥14入')) return { spec:'14', name:'鳳梨酥 14入', price:600, mix:false, flavor1:null, flavor2:null, qty }; const mix = line.includes('綜合搭配'); const is6 = line.includes('6入'); const spec = is6 ? '6' : '8'; const name = is6 ? '月餅 6入' : '月餅 8入'; const price = is6 ? 450 : 580; if (mix) return { spec, name, price, mix:true, flavor1:null, flavor2:null, qty }; // 解析口味 const halfMatch = line.match(/(.+)＆(.+)各半/); if (halfMatch) return { spec, name, price, mix:false, flavor1:halfMatch[1].replace(name,'').replace(/×\d+/,'').trim(), flavor2:halfMatch[2].trim(), qty }; const soloMatch = line.match(/(.+)整盒/); const f1 = soloMatch ? soloMatch[1].replace(name,'').replace(/×\d+/,'').trim() : null; return { spec, name, price, mix:false, flavor1:f1, flavor2:null, qty }; } // ── POST：新增/更新/刪除 ── function doPost(e) { try { const raw = e.parameter.payload || e.postData.contents; const data = JSON.parse(raw); const action = data.action || 'add'; if (action === 'add') return addOrder(data); if (action === 'update') return updateOrder(data); if (action === 'delete') return softDeleteOrder(data); return jsonResponse({ success: false, error: '未知 action' }); } catch(err) { return jsonResponse({ success: false, error: err.message }); } } // ── 新增訂單 ── function addOrder(order) { const sheet = getOrCreateSheet(); const itemLines = order.items.map(it => { const q = it.qty||1; const qs = q>1 ? ` ×${q}` : ''; if (it.spec === '14') return `鳳梨酥14入${qs}`; if (it.mix) return `${it.name}${qs} 綜合搭配`; if (it.flavor1 && it.flavor2) return `${it.name}${qs} ${it.flavor1}＆${it.flavor2}各半`; return `${it.name}${qs} ${it.flavor1||''}整盒`; }).join('\n'); const row = [ order.id, order.name, order.date, order.pay === 'paid' ? '已付款' : '未付款', order.shipped ? '已出貨' : '待出貨', order.total, itemLines, order.note || '', new Date(order.createdAt).toLocaleString('zh-TW'), false, ]; sheet.appendRow(row); const lastRow = sheet.getLastRow(); sheet.getRange(lastRow, 1, 1, HEADERS.length) .setBorder(true,true,true,true,true,true); return jsonResponse({ success: true }); } // ── 更新狀態 ── function updateOrder(data) { const sheet = getOrCreateSheet(); const id = String(data.id); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === id) { if (data.pay !== undefined) sheet.getRange(i+1, 4).setValue(data.pay === 'paid' ? '已付款' : '未付款'); if (data.shipped !== undefined) sheet.getRange(i+1, 5).setValue(data.shipped ? '已出貨' : '待出貨'); return jsonResponse({ success: true }); } } return jsonResponse({ success: false, error: '找不到訂單' }); } // ── 軟刪除（標記，不真的刪列） ── function softDeleteOrder(data) { const sheet = getOrCreateSheet(); const id = String(data.id); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === id) { sheet.getRange(i+1, 10).setValue('已刪除'); // 整列變灰 sheet.getRange(i+1, 1, 1, HEADERS.length) .setFontColor('#AAAAAA') .setBackground('#F5F5F5'); return jsonResponse({ success: true }); } } return jsonResponse({ success: false, error: '找不到訂單' }); } // ── 建立工作表 ── function getOrCreateSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); sheet.getRange(1, 1, 1, HEADERS.length) .setBackground('#7A4A14').setFontColor('#FAF5EC') .setFontWeight('bold').setFontSize(11) .setHorizontalAlignment('center'); [120,100,100,90,90,80,260,180,160,80].forEach((w,i)=>sheet.setColumnWidth(i+1,w)); sheet.setFrozenRows(1); } return sheet; } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
